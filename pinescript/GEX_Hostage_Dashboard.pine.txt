getLabelCoordinates(position, xOff, yOffATR, atrPad) =>
    int rightEdge = bar_index + 1
    int x = na
    float y = na

    // X coordinate (bars)
    if position == "Top Left" or position == "Middle Left" or position == "Bottom Left"
        x := bar_index - xOff
    else if position == "Top Center" or position == "Middle Center" or position == "Bottom Center"
        x := int(math.avg(bar_index, rightEdge))
    else
        x := rightEdge + xOff

    // Y coordinate (ATR padded)
    pad = ta.atr(14) * atrPad
    if position == "Top Left" or position == "Top Center" or position == "Top Right"
        y := high + pad + (ta.atr(14) * yOffATR)
    else if position == "Middle Left" or position == "Middle Center" or position == "Middle Right"
        y := math.avg(high, low) + (ta.atr(14) * yOffATR)
    else
        y := low - pad - (ta.atr(14) * yOffATR)

    [x, y]

// REPLAY-STABLE ORB STATE MACHINE (for use inside request.security per symbol)
// Returns: orh, orl, or_complete, holds, failed, dirTxt
// ═══════════════════════════════════════════════════════════════════════════════
f_orb_state(_or_minutes, _hold_bars, _use_vwap) =>
    rthSess = "0930-1600"
    in_sess = not na(time(timeframe.period, rthSess))
    is_new  = ta.change(time("D")) != 0

    var float orh = na
    var float orl = na
    var bool  or_complete = false
    var int   bars_in_or = 0

    var int   hold_count = 0
    var int   last_break_dir = 0   // +1 long, -1 short, 0 none
    var int   break_bars_ago = na

    if is_new
        orh := na
        orl := na
        or_complete := false
        bars_in_or := 0
        hold_count := 0
        last_break_dir := 0
        break_bars_ago := na

    is_first = in_sess and not in_sess[1]
    if is_first
        orh := high
        orl := low
        or_complete := false
        bars_in_or := 0
        hold_count := 0
        last_break_dir := 0
        break_bars_ago := na

    if in_sess and not or_complete
        bars_in_or += 1
        mins_elapsed = bars_in_or * timeframe.in_seconds(timeframe.period) / 60.0

        if mins_elapsed < _or_minutes
            orh := math.max(orh, high)
            orl := math.min(orl, low)
        else
            or_complete := true

    v = ta.vwap(close)
    vwap_ok_long  = (not _use_vwap) or close > v
    vwap_ok_short = (not _use_vwap) or close < v

    above_or  = or_complete and not na(orh) and close > orh
    below_or  = or_complete and not na(orl) and close < orl
    inside_or = or_complete and not above_or and not below_or

    break_bars_ago := na(break_bars_ago) ? na : break_bars_ago
    if or_complete
        if (above_or and last_break_dir != 1)
            last_break_dir := 1
            hold_count := 0
            break_bars_ago := 0
        else if (below_or and last_break_dir != -1)
            last_break_dir := -1
            hold_count := 0
            break_bars_ago := 0
        else if not na(break_bars_ago)
            break_bars_ago += 1

    if or_complete
        if last_break_dir == 1
            if close > orh and vwap_ok_long
                hold_count += 1
            else
                hold_count := 0
        else if last_break_dir == -1
            if close < orl and vwap_ok_short
                hold_count += 1
            else
                hold_count := 0
        else
            hold_count := 0

    holds = or_complete and hold_count >= _hold_bars

    failed = or_complete and inside_or and not na(break_bars_ago) and break_bars_ago <= _hold_bars and (last_break_dir != 0) and not holds

    dirTxt = holds and last_break_dir == 1 ? "UP" :
             holds and last_break_dir == -1 ? "DOWN" :
             (last_break_dir == 1 and above_or) ? "TRY UP" :
             (last_break_dir == -1 and below_or) ? "TRY DN" : "—"

    [orh, orl, or_complete, holds, failed, dirTxt]
